  \item \cppcode{move()}.
  \item \cppcode{swap_ranges()}.
  \item \cppcode{replace}, \cppcode{replace_if()}, \cppcode{replace_copy()}, \cppcode{replace_copy_if()}.
  \item \cppcode{fill()}, \cppcode{fill_n()}, \cppcode{generate()}, \cppcode{generate_n()}.
  \item \cppcode{remove()}, \cppcode{remove_if()}, \cppcode{remove_copy()}, \cppcode{remove_copy_if()}.
  \item \cppcode{unique()}, \cppcode{unique_copy()}.
  \item \cppcode{reverse()}, \cppcode{reverse_copy()}.
  \item \cppcode{rotate()}, \cppcode{rotate_copy()}, \cppcode{shift_left()}, \cppcode{shift_right()},
